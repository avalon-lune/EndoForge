cmake_minimum_required(VERSION 3.24)
project(LapPhysics LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(SOFA_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../ThirdParty/SOFA/sdk/SOFA_v26.06.00_Win64" CACHE PATH "SOFA SDK")
set(EIGEN3_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../ThirdParty/physics-tools/eigen-3.4.0" CACHE PATH "Eigen headers")
set(BOOST_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../ThirdParty/physics-tools/boost_1_87_0" CACHE PATH "Boost headers")
if(POLICY CMP0167)
  cmake_policy(SET CMP0167 OLD)
endif()
list(APPEND CMAKE_PREFIX_PATH "${SOFA_ROOT}")
find_package(Sofa.Simulation.Graph REQUIRED)
find_package(Sofa.Component.StateContainer REQUIRED)
find_package(Sofa.Component.Collision.Response.Contact REQUIRED)
find_package(Sofa.Component.Collision.Detection.Intersection REQUIRED)
find_package(Sofa.Component.Constraint.Lagrangian.Solver REQUIRED)
find_package(Sofa.Component.SolidMechanics.FEM.Elastic REQUIRED)
add_executable(LapPhysics main.cpp)
target_include_directories(LapPhysics PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../ThirdParty/physics-tools")
target_compile_definitions(LapPhysics PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(LapPhysics PRIVATE /utf-8 /W3)
target_link_libraries(LapPhysics PRIVATE Sofa.Simulation.Graph Sofa.Component.StateContainer Sofa.Component.Collision.Response.Contact Sofa.Component.Collision.Detection.Intersection Sofa.Component.Constraint.Lagrangian.Solver Sofa.Component.SolidMechanics.FEM.Elastic ws2_32 winmm)
set_target_properties(LapPhysics PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_SOURCE_DIR}/runtime")

# Standalone geometry/continuous-admission tests do not need a SOFA graph.
add_executable(Phase6ToolCollisionTests EXCLUDE_FROM_ALL Phase6ToolCollisionTests.cpp)
target_compile_options(Phase6ToolCollisionTests PRIVATE /utf-8 /W4 /O2)
enable_testing()
add_test(NAME Phase6ToolCollision COMMAND Phase6ToolCollisionTests)
